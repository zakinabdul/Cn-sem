Q1TCP

1. Factorial of a number using TCP

The client sends an integer N. The server calculates N! and returns it. This version accepts 0 to 20 so the result fits in unsigned long long.

Port: 8081 · Files: server1.c and client1.c

Server program — server1.c

// server1.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>

int main() {
    int s, c, n, i;
    unsigned long long fact = 1;
    char b[100];
    struct sockaddr_in server;

    s = socket(AF_INET, SOCK_STREAM, 0);

    server.sin_family = AF_INET;
    server.sin_port = htons(8081);
    server.sin_addr.s_addr = INADDR_ANY;

    bind(s, (struct sockaddr *)&server, sizeof(server));
    listen(s, 1);
    printf("Waiting for client...\n");

    c = accept(s, NULL, NULL);
    recv(c, b, sizeof(b), 0);
    n = atoi(b);

    if (n < 0 || n > 20) {
        strcpy(b, "Enter N between 0 and 20");
    } else {
        for (i = 1; i <= n; i++)
            fact = fact * i;
        sprintf(b, "%llu", fact);
    }

    send(c, b, strlen(b) + 1, 0);
    close(c);
    close(s);
    return 0;
}

Client program — client1.c

// client1.c
#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>

int main() {
    int s;
    char b[100];
    struct sockaddr_in server;

    s = socket(AF_INET, SOCK_STREAM, 0);

    server.sin_family = AF_INET;
    server.sin_port = htons(8081);
    server.sin_addr.s_addr = inet_addr("127.0.0.1");

    connect(s, (struct sockaddr *)&server, sizeof(server));

    printf("Enter a number (0 to 20): ");
    fgets(b, sizeof(b), stdin);
    b[strcspn(b, "\n")] = '\0';

    send(s, b, strlen(b) + 1, 0);
    recv(s, b, sizeof(b), 0);

    printf("Factorial = %s\n", b);
    close(s);
    return 0;
}

4. Length of a string using TCP

The client sends a string. The server calculates strlen() and sends the length back. Spaces are allowed.

Port: 8084 · Files: server4.c and client4.c

Server program — server4.c

// server4.c
#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>

int main() {
    int s, c, length;
    char b[1024];
    struct sockaddr_in server;

    s = socket(AF_INET, SOCK_STREAM, 0);
    server.sin_family = AF_INET;
    server.sin_port = htons(8084);
    server.sin_addr.s_addr = INADDR_ANY;

    bind(s, (struct sockaddr *)&server, sizeof(server));
    listen(s, 1);
    printf("Waiting for client...\n");

    c = accept(s, NULL, NULL);
    recv(c, b, sizeof(b), 0);
    b[sizeof(b) - 1] = '\0';

    length = strlen(b);
    sprintf(b, "%d", length);

    send(c, b, strlen(b) + 1, 0);
    close(c);
    close(s);
    return 0;
}

Part 2 — UDP programs

UDP uses SOCK_DGRAM, bind(), sendto() and recvfrom(). There is no listen() or accept().

Q5UDP

5. Factorial of a number using UDP

The client sends N in a UDP datagram. The server calculates N! and sends the answer to the client's address. Accepts 0 to 20.

Port: 8085 · Files: server5.c and client5.c

Server program — server5.c

// server5.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>

int main() {
    int s, n, i;
    unsigned long long fact = 1;
    char b[100];
    struct sockaddr_in server, client;
    socklen_t len = sizeof(client);

    s = socket(AF_INET, SOCK_DGRAM, 0);
    server.sin_family = AF_INET;
    server.sin_port = htons(8085);
    server.sin_addr.s_addr = INADDR_ANY;

    bind(s, (struct sockaddr *)&server, sizeof(server));
    printf("Waiting for datagram...\n");

    recvfrom(s, b, sizeof(b) - 1, 0,
             (struct sockaddr *)&client, &len);
    b[sizeof(b) - 1] = '\0';
    n = atoi(b);

    if (n < 0 || n > 20) {
        strcpy(b, "Enter N between 0 and 20");
    } else {
        for (i = 1; i <= n; i++)
            fact = fact * i;
        sprintf(b, "%llu", fact);
    }

    sendto(s, b, strlen(b) + 1, 0,
           (struct sockaddr *)&client, len);
    close(s);
    return 0;
}

Q8UDP

8. Length of a string using UDP

The client sends a string datagram. The server uses strlen() and returns the length. Spaces are allowed.

Port: 8088 · Files: server8.c and client8.c

Server program — server8.c

// server8.c
#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>

int main() {
    int s, length, received;
    char b[1024];
    struct sockaddr_in server, client;
    socklen_t len = sizeof(client);

    s = socket(AF_INET, SOCK_DGRAM, 0);
    server.sin_family = AF_INET;
    server.sin_port = htons(8088);
    server.sin_addr.s_addr = INADDR_ANY;

    bind(s, (struct sockaddr *)&server, sizeof(server));
    printf("Waiting for datagram...\n");

    received = recvfrom(s, b, sizeof(b) - 1, 0,
                        (struct sockaddr *)&client, &len);
    if (received < 0) {
        close(s);
        return 1;
    }
    b[received] = '\0';

    length = strlen(b);
    sprintf(b, "%d", length);

    sendto(s, b, strlen(b) + 1, 0,
           (struct sockaddr *)&client, len);
    close(s);
    return 0;
}

Quick revision — TCP vs UDP

TCPUDP
SOCK_STREAMSOCK_DGRAM
Connection-orientedConnectionless
listen() and accept()No listen() or accept()
connect() on clientNo connect() needed here
send() and recv()sendto() and recvfrom()
Reliable, ordered byte streamDatagrams; delivery/order not guaranteed

Common socket functions

  • socket() — creates a socket.
  • bind() — assigns the server's IP address and port.
  • listen() — makes a TCP socket wait for connections.
  • accept() — accepts a TCP client.
  • connect() — connects a TCP client to the server.
  • send()/recv() — sends and receives TCP data.
  • sendto()/recvfrom() — sends and receives UDP datagrams.
  • close() — closes a socket.
